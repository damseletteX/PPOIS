60void Set::parse(const std::string &s)
61{
62 const char *bad = "Invalid set string: check the brackets and stray characters.";
63 std::stack<Set> lvls;
64 std::string curElement;
65 Set ready;
66 Set result;
67 for (char c : s)
68 {
69 switch (c)
70 {
71 case ' ':
72 case '\t':
73 case '\n':
74 case '\r':
75 break;
76 case '{':
77 lvls.push(Set());
78 break;
79 case '}':
80 if (lvls.empty())
81 throw std::invalid_argument(bad);
82 if (!curElement.empty())
83 lvls.top().insertUnique(Element(curElement));
84 curElement.clear();
85 ready = lvls.top();
86 lvls.pop();
87 if (lvls.empty())
88 {
89 result = ready;
90 }
91 else
92 {
93 lvls.top().insertUnique(Element(ready));
94 }
95 break;
96 case ',':
97 if (lvls.empty())
98 throw std::invalid_argument(bad);
99 if (!curElement.empty())
100 {
101 lvls.top().insertUnique(Element(curElement));
102 curElement.clear();
103 }
104 break;
105 default:
106 if (lvls.empty())
107 throw std::invalid_argument(bad);
108 curElement += c;
109 break;
110 }
111 }
112
113 if (!lvls.empty())
114 throw std::invalid_argument(bad);
115 *this = result;
116}
